import fpinscala.datastructures._
import fpinscala.datastructures.List._

// =============================================================================
// Tests for sum3
// =============================================================================

// Empty list
assert(sum3(Nil) == 0)

// Single element & negatives
assert(sum3(List(5)) == 5)
assert(sum3(List(-5, 5)) == 0)

// Multi-element
assert(sum3(List(1, 2, 3, 4)) == 10)

// =============================================================================
// Tests for product3
// =============================================================================

// Empty list (neutral element for multiplication)
assert(product3(Nil) == 1.0)

// Single element
assert(product3(List(3.0)) == 3.0)

// Multi-element
assert(product3(List(1.0, 2.0, 3.0, 4.0)) == 24.0)

// Multiplication by zero
assert(product3(List(2.0, 0.0, 5.0)) == 0.0)

// =============================================================================
// Tests for length2
// =============================================================================

// Empty list
assert(length2(Nil) == 0)
// Single element
assert(length2(List("scala")) == 1)

// Multi-element of different types
assert(length2(List(1, 2, 3, 4, 5)) == 5)
assert(length2(List(true, false, true)) == 3)

// =============================================================================
// Stack Safety Check (verifying tail recursion via foldLeft)
// =============================================================================

// Suppress output with '; ()' to avoid REPL toString recursion crash:
lazy val largeList: List[Int] = (1 to 10000).foldRight(Nil: List[Int])(Cons(_, _))

// All three should run safely in constant stack space:
assert(sum3(largeList) == 50005000)
assert(length2(largeList) == 10000)
