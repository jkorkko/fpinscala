// --- Test Case 1: Empty list (Nil) ---
assert(revert(Nil) == Nil)

// --- Test Case 2: Single element (Identity) ---
assert(revert(List(42)) == List(42))
assert(revert(List("single")) == List("single"))

// --- Test Case 3: Standard multi-element list ---
val nums = List(1, 2, 3, 4, 5)
assert(revert(nums) == List(5, 4, 3, 2, 1))

val words = List("first", "second", "third")
assert(revert(words) == List("third", "second", "first"))

// --- Test Case 4: Double reverse (Invariance law) ---
// For any list xs, revert(reverse(xs)) == xs
val original = List(10, 20, 30)
assert(revert(revert(original)) == original)

// --- Test Case 5: Palindrome preservation ---
val palindrome = List(1, 2, 3, 2, 1)
assert(revert(palindrome) == palindrome)
