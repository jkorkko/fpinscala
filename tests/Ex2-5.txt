import fpinscala.gettingstarted.PolymorphicFunctions._

val double = (x: Int) => x * 2
val addThree = (x: Int) => x + 3
val addThenDouble = compose(double, addThree)
assert(addThenDouble(2) == 10)

val numToString = (n: Int) => s"Item-$n"
val isLong = (s: String) => s.length > 6
val checkId = compose(isLong, numToString)
assert(checkId(5) == false)
assert(checkId(100) == true)

val id = (x: Int) => x
val square = (x: Int) => x * x
val squareWithIdLeft = compose(square, id)
val squareWithIdRight = compose(id, square)
assert(squareWithIdLeft(7) == 49)
assert(squareWithIdRight(7) == 49)
